1"""Protocol: the contract between the core and one bus type.
2
3The core does not know how a protocol works inside. It gives the protocol the bytes to send and gets
4back the UTS request body; it gives the protocol an item UTS returned and gets back the bytes received.
5
6    TX:  encode(data)    -> body for UTS (POST on path / interface)
7    RX:  decode(item)    -> Received(data, timestamp)
8    simulate(data, ts)   -> the item UTS would return (only for the fake UTS in tests)
9
10Step 1 (Ethernet): bytes only. Addresses (1553 node / subnet, CAN id) and protocol settings come in
11later steps, when a protocol needs them.
12"""
13from abc import ABC, abstractmethod
14from dataclasses import dataclass
15from typing import Optional
16
17
18@dataclass(frozen=True)
19class Received:
20    data: bytes                         # the bytes on the line (header included, as before)
21    timestamp: Optional[float]          # UTS arrival time in seconds
22
23
24class Protocol(ABC):
25    path: str = ""                      # bus name in the UTS URL: /{path}/{interface}/messages
26
27    @abstractmethod
28    def encode(self, data: bytes) -> dict:
29        """TX: bytes on the line -> the UTS request body."""
30
31    @abstractmethod
32    def decode(self, item: dict) -> Received:
33        """RX: one item of the UTS answer -> the bytes received and their time."""
34
35    def simulate(self, data: bytes, timestamp: Optional[float]) -> dict:
36        """The item UTS would return if the device sent data (for the fake UTS). Optional."""
37        raise NotImplementedError(f"{type(self).__name__} cannot be simulated")

21        return {"message": list(data)}

272        Body bytes go to the start of the buffer, trailer bytes to the checksum field;
273        the unused part of a variable-size message is cleared.
274        timestamp: arrival time reported by UTS (seconds), stored in rx_state.timestamp.

279            return False

282        ctypes.memset(address, 0, ctypes.sizeof(self))
283        ctypes.memmove(address, raw, body)
284        ctypes.memmove(address + offset, raw[body:body + size], size)

24    def __init__(self, transport: Transport, bus: EventBus, log=None):
25        self.transport = transport
26        self.bus = bus
27        self.log = log or logging.getLogger(__name__)
28        self._send_lock = threading.Lock()
29        # id(message) -> [message, next due (monotonic)]; ctypes objects are not hashable
30        self._table: dict[int, list] = {}
31        self._wake = threading.Event()
32        self._stop = threading.Event()
33        self._thread: Optional[threading.Thread] = None
34
35    # ---------------------------------------------------------------- sending
36
37    def send(self, msg: BaseMessageStructure) -> bool:
38        """Sends a message once. Returns False if the UTS request failed (LinkError is published)."""
39        with self._send_lock:
40            raw = msg.pack()
41            image = bytes(msg)
42            try:
43                wire = self.transport.send(msg._channel, raw)
44            except Exception as error:
45                self.log.exception("UTS send of %s failed on %s", type(msg).__name__, msg._channel)
46                self.bus.publish(LinkError(msg._channel, "send", error, time.time()))
47                return False
48            msg._sent()
49        self.log.debug("TX %s on %s (%d bytes)", type(msg).__name__, msg._channel, len(raw))
50        now = time.time()
51        self.bus.publish(WireSent(msg._channel, msg, wire, now))
52        self.bus.publish(MessageSent(msg, raw, now, image))
53        return True
54
55    # ---------------------------------------------------------------- periodic
56
57    def start(self, msg: BaseMessageStructure) -> None:
58        if msg._tx is None or not msg._tx.timing.is_periodic:
59            raise ValueError(f"{type(msg).__name__} has no periodic TX timing (tx=Timing(period))")
60        self._table[id(msg)] = [msg, time.monotonic()]
61        self.log.info("periodic TX started: %s every %s s", type(msg).__name__, msg._tx.timing.period)
62        self._wake.set()
63
64    def stop(self, msg: BaseMessageStructure) -> None:
65        if self._table.pop(id(msg), None) is not None:
66            self.log.info("periodic TX stopped: %s", type(msg).__name__)
67        self._wake.set()
68
69    def is_running(self, msg: BaseMessageStructure) -> bool:
70        return id(msg) in self._table
71
72    def tick(self) -> Optional[float]:
73        """Sends what is due; returns seconds until the next due message (None if the table is empty)."""
74        now = time.monotonic()
75        for key, entry in list(self._table.items()):
76            msg, due = entry
77            if not msg._registered:
78                self._table.pop(key, None)
79                continue
80            if due <= now:
81                self.send(msg)
82                entry[1] = due + msg._tx.timing.period   # harmless if stop() removed the entry meanwhile
83        if not self._table:
84            return None
85        return max(0.0, min(due for _, due in list(self._table.values())) - time.monotonic())
86
87    # ---------------------------------------------------------------- thread
88
89    def start_thread(self) -> None:
90        self._stop.clear()
91        self._thread = threading.Thread(target=self._run, name="TxScheduler", daemon=True)
92        self._thread.start()
93
94    def stop_thread(self) -> None:
95        self._stop.set()
96        self._wake.set()
97        if self._thread is not None:
98            self._thread.join()
99
100    def _run(self) -> None:
101        while not self._stop.is_set():
102            wait = self.tick()
103            self._wake.wait(wait)
104            self._wake.clear()

1"""TxScheduler: one-shot sends and one thread for all periodic messages.
2
3- send(msg): counter, checksum, bytes, envelope and the UTS request happen under one lock, so the
4  scheduler and the test sending the same message at the same time cannot mix up the counter.
5  Publishes WireSent (bytes on the line) and MessageSent. After a successful send the after_send
6  hooks of the special fields run (still under the lock); the events carry the values that were sent.
7- start(msg) / stop(msg): adds / removes a periodic message. The next send time is advanced by
8  the period (not now + period), so the delay does not accumulate.
9- A message that is no longer registered (replaced by a new object) is dropped from the table.
10Logs: a failed send as ERROR (with traceback), periodic start / stop as INFO, every send as DEBUG.
11"""
